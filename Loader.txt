repeat task.wait() until game:IsLoaded()

local MarketplaceService = game:GetService("MarketplaceService")
local PlaceId = game.PlaceId

    }
}

local SelectedGame = nil

--// METHOD 1: PLACE ID DETECTION
for _, data in ipairs(Games) do
    for _, id in ipairs(data.PlaceIds) do
        if PlaceId == id then
            SelectedGame = data
            break
        end
    end

    if SelectedGame then break end
end

--// METHOD 2: GAME NAME DETECTION
if not SelectedGame then
    local Success, Info = pcall(function()
        return MarketplaceService:GetProductInfo(PlaceId)
    end)

    if Success and Info and Info.Name then
        local GameName = string.lower(Info.Name)

        if string.find(GameName, "blox fruits", 1, true) then
            if string.find(GameName, "sea 3", 1, true) then
                SelectedGame = Games[4]
            elseif string.find(GameName, "sea 2", 1, true) then
                SelectedGame = Games[3]
            elseif string.find(GameName, "sea 1", 1, true) then
                SelectedGame = Games[2]
            else
                SelectedGame = Games[2]
            end
        else
            for _, data in ipairs(Games) do
                for _, alias in ipairs(data.Aliases) do
                    if string.find(GameName, string.lower(alias), 1, true) then
                        SelectedGame = data
                        break
                    end
                end

                if SelectedGame then break end
            end
        end
    end
end

--// UNSUPPORTED GAME
if not SelectedGame then
    warn("[ROXYS HUB] Unsupported Game: " .. tostring(PlaceId))
    return
end

print("[ROXYS HUB] Detected: " .. SelectedGame.Name)
print("[ROXYS HUB] Loading script...")

--// EXECUTION
local Success, Error = pcall(function()
    local Source = game:HttpGet(SelectedGame.URL)

    assert(type(Source) == "string" and #Source > 0, "Empty script response")

    local Script = loadstring(Source)
    assert(Script, "Compilation failed")

    Script()
end)

if not Success then
    warn("[ROXYS HUB] Error: " .. tostring(Error))
else
    print("[ROXYS HUB] Successfully loaded " .. SelectedGame.Name)
end
